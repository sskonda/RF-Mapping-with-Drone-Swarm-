cmake_minimum_required(VERSION 3.16)
project(flight_bench_tests C)
enable_testing()
option(FC_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
set(FC_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../ESP32_Code/Autonomy/Flight_Controller/main")
add_compile_options(-std=c11 -O2 -g -Wall -Wextra -Werror -Wpedantic -Wshadow -UNDEBUG)
if(FC_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_executable(test_sensors test_sensors.c "${FC_SOURCE}/sensor_decode.c")
add_executable(test_bench test_bench.c "${FC_SOURCE}/bench.c")
foreach(name sensors bench)
    target_include_directories(test_${name} PRIVATE "${FC_SOURCE}")
    target_link_libraries(test_${name} m)
    add_test(NAME ${name} COMMAND test_${name})
endforeach()
